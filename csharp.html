Excel VSTO Add-in：動態讀取設定檔產生 Ribbon 選單

本教學說明如何在 C# Excel VSTO 專案中，讀取指定的 Config 文字檔（如 X 槽），並根據內容動態產生對應數量的群組（Group）、下拉選單（DropDown）與按鈕。

核心概念： 由於 Visual Studio 的「視覺化設計工具 (Ribbon Designer)」無法動態增減群組，我們必須改用 Ribbon (XML) 的方式，透過 C# 程式碼動態組裝 XML 字串來定義介面。

步驟 1：準備設定檔

請在您的測試路徑（例如 X:\Data\Config.txt）建立以下格式的文字檔。請確保編碼為 UTF-8。

檔案內容範例：Config.txt 
[Stage1]
func1
func2
func3

[Stage2]
func4
func5
func6

[Stage3]
func7
func8
func9

步驟 2：建立 VSTO 專案並加入 Ribbon XML

  1. 開啟 Visual Studio，建立一個新的 Excel VSTO Add-in 專案。
  2. 在「方案總管」專案名稱上按右鍵 → 加入 → 新增項目。
  3. 選擇 功能區 (XML) (Ribbon (XML))。
  4. 將其命名為 DynamicRibbon.cs。
  5. 注意：系統會產生 .cs 與 .xml 檔。請忽略並不要編輯那個 .xml 檔，因為我們將在程式碼中動態生成 XML。

步驟 3：實作核心邏輯代碼

請將 DynamicRibbon.cs 的內容完全替換為以下程式碼。這包含了資料模型、XML 建構器以及事件處理邏輯。

檔案：DynamicRibbon.cs 
using System;
using System.Collections.Generic;
using System.IO;
using System.Runtime.InteropServices;
using System.Text;
using System.Windows.Forms;
using Microsoft.Office.Core;
using Excel = Microsoft.Office.Interop.Excel;

namespace YourProjectName // ★請確認這裡的 namespace 與你的專案一致
{
    // 定義資料結構
    public class StageGroup
    {
        public string GroupName { get; set; }
        public List<string> Macros { get; set; }
        public string SelectedMacro { get; set; }

        public StageGroup()
        {
            Macros = new List<string>();
        }
    }

    [ComVisible(true)]
    public class DynamicRibbon : IRibbonExtensibility
    {
        private IRibbonUI _ribbon;
        private List<StageGroup> _stages;
        
        // 設定檔路徑 (建議使用 UNC 路徑避免磁碟機未掛載問題)
        private const string FilePath = @"X:\Data\Config.txt"; 

        public DynamicRibbon()
        {
            _stages = new List<StageGroup>();
        }

        #region IRibbonExtensibility 成員

        // 這是最重要的方法：Excel 啟動時會呼叫此方法取得介面定義 XML
        public string GetCustomUI(string ribbonID)
        {
            // 1. 讀取設定檔資料
            LoadDataFromFile();

            // 2. 開始組裝 XML 字串
            StringBuilder xml = new StringBuilder();
            xml.Append(@"<customUI xmlns=""http://schemas.microsoft.com/office/2009/07/customui"" onLoad=""Ribbon_Load"">");
            xml.Append(@"  <ribbon>");
            xml.Append(@"    <tabs>");
            xml.Append(@"      <tab id=""tabCustom"" label=""自動化工具"">");

            // 3. 根據讀到的 Stage 數量動態產生 Group
            for (int i = 0; i < _stages.Count; i++)
            {
                var stage = _stages[i];
                string groupId = $"group_{i}";
                string ddId = $"dd_{i}";
                string btnExecId = $"btnExec_{i}";
                string btnRefreshId = "btnRefresh";

                // 建立 Group
                xml.Append($@"<group id=""{groupId}"" label=""{stage.GroupName}"">");

                // 3.1 僅在第一個 Group 加入 Refresh 按鈕
                if (i == 0)
                {
                    xml.Append($@"<button id=""{btnRefreshId}"" label=""重新整理"" size=""large"" imageMso=""Refresh"" onAction=""OnRefreshClick""/>");
                    xml.Append(@"<separator id=""sep1"" />");
                }

                // 3.2 加入下拉選單 (使用 Callbacks 動態給值)
                // tag 屬性存 index，方便事件觸發時知道是第幾個 group
                xml.Append($@"<dropDown id=""{ddId}"" label=""選擇功能"" tag=""{i}"" 
                                getItemCount=""GetItemCount"" 
                                getItemLabel=""GetItemLabel"" 
                                getSelectedItemIndex=""GetSelectedItemIndex""
                                onAction=""OnDropDownChange""/>");

                // 3.3 加入執行按鈕
                xml.Append($@"<button id=""{btnExecId}"" label=""Execute"" tag=""{i}"" imageMso=""PlayMacro"" onAction=""OnExecuteClick""/>");

                xml.Append(@"</group>");
            }

            xml.Append(@"      </tab>");
            xml.Append(@"    </tabs>");
            xml.Append(@"  </ribbon>");
            xml.Append(@"</customUI>");

            return xml.ToString();
        }

        #endregion

        #region 檔案讀取邏輯
        private void LoadDataFromFile()
        {
            _stages.Clear();
            if (!File.Exists(FilePath))
            {
                var errorGroup = new StageGroup { GroupName = "錯誤" };
                errorGroup.Macros.Add("找不到設定檔");
                _stages.Add(errorGroup);
                return;
            }

            string[] lines = File.ReadAllLines(FilePath, Encoding.UTF8);
            StageGroup currentGroup = null;

            foreach (string line in lines)
            {
                string text = line.Trim();
                if (string.IsNullOrEmpty(text)) continue;

                if (text.StartsWith("[") && text.EndsWith("]"))
                {
                    currentGroup = new StageGroup();
                    currentGroup.GroupName = text.Trim('[', ']');
                    _stages.Add(currentGroup);
                }
                else
                {
                    if (currentGroup != null)
                    {
                        currentGroup.Macros.Add(text);
                        // 預設選取第一個
                        if (string.IsNullOrEmpty(currentGroup.SelectedMacro))
                            currentGroup.SelectedMacro = text;
                    }
                }
            }
        }
        #endregion

        #region Ribbon Callbacks (事件處理)

        public void Ribbon_Load(IRibbonUI ribbonUI)
        {
            this._ribbon = ribbonUI;
        }

        // 下拉選單：總共有幾個項目？
        public int GetItemCount(IRibbonControl control)
        {
            int index = int.Parse(control.Tag); 
            if (index < _stages.Count)
                return _stages[index].Macros.Count;
            return 0;
        }

        // 下拉選單：取得第 itemIndex 個項目的文字
        public string GetItemLabel(IRibbonControl control, int itemIndex)
        {
            int groupIndex = int.Parse(control.Tag);
            return _stages[groupIndex].Macros[itemIndex];
        }

        // 下拉選單：取得目前選中的索引
        public int GetSelectedItemIndex(IRibbonControl control)
        {
            int groupIndex = int.Parse(control.Tag);
            var group = _stages[groupIndex];
            return group.Macros.IndexOf(group.SelectedMacro);
        }

        // 當使用者改變下拉選單時
        public void OnDropDownChange(IRibbonControl control, string selectedId, int selectedIndex)
        {
            int groupIndex = int.Parse(control.Tag);
            _stages[groupIndex].SelectedMacro = _stages[groupIndex].Macros[selectedIndex];
        }

        // 執行按鈕 (Execute)
        public void OnExecuteClick(IRibbonControl control)
        {
            int groupIndex = int.Parse(control.Tag);
            string macroName = _stages[groupIndex].SelectedMacro;

            if (string.IsNullOrEmpty(macroName)) return;

            try
            {
                // 執行 Excel 巨集
                Globals.ThisAddIn.Application.Run(macroName);
            }
            catch (Exception ex)
            {
                MessageBox.Show($"執行 '{macroName}' 失敗: {ex.Message}");
            }
        }

        // 重新整理按鈕 (Refresh)
        public void OnRefreshClick(IRibbonControl control)
        {
            int oldStageCount = _stages.Count;
            LoadDataFromFile(); // 重新讀檔

            if (_stages.Count != oldStageCount)
            {
                MessageBox.Show("偵測到 Stage 數量變更，請重新啟動 Excel 以套用新的介面結構。", "提示");
            }
            else
            {
                // 若數量沒變，僅重新繪製內容 (更新下拉選單項目)
                _ribbon.Invalidate(); 
                MessageBox.Show("設定已更新！");
            }
        }
        #endregion
    }
}

步驟 4：掛載 Ribbon

最後，必須告訴 Excel 使用我們自訂的 Ribbon 類別。請打開 ThisAddIn.cs，並加入 override 方法。

檔案：ThisAddIn.cs 
public partial class ThisAddIn
{
    private void ThisAddIn_Startup(object sender, System.EventArgs e)
    {
    }

    private void ThisAddIn_Shutdown(object sender, System.EventArgs e)
    {
    }

    // ★加入這段程式碼來覆寫預設的 Ribbon 載入行為
    protected override Microsoft.Office.Core.IRibbonExtensibility CreateRibbonExtensibilityObject()
    {
        return new DynamicRibbon();
    }

    #region VSTO 產生的程式碼
    private void InternalStartup()
    {
        this.Startup += new System.EventHandler(ThisAddIn_Startup);
        this.Shutdown += new System.EventHandler(ThisAddIn_Shutdown);
    }
    #endregion
}

功能說明與限制

  • 動態生成： 程式啟動時會解析 GetCustomUI 中的 XML 迴圈，若設定檔有 3 個 Stage，就會產生 3 個 Group。
  • Tag 屬性： 我們在 XML 中使用 tag="{i}" 來標記每個控制項屬於第幾個 Group，這樣所有按鈕共用同一個 OnExecuteClick 事件時，就能分辨要執行哪一個 Group 的巨集。
  • 重新整理 (Refresh)：
    • 若只是修改下拉選單的內容（例如 func1 改成 func_new），點擊 Refresh 即可立即更新。
    • 若新增或刪除 Stage（改變 Group 數量），由於 Ribbon XML 架構是在 Excel 啟動時決定的，必須關閉並重啟 Excel 才會生效。
